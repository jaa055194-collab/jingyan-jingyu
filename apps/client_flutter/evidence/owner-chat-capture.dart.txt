import 'dart:io';
import 'dart:ui' as ui;

import 'package:client_flutter/app/shell_app.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:material_ui/material_ui.dart';

import 'owner_chat_entry_test.dart' show OwnerChatFixture, entry, directId;

void main() {
  testWidgets('renders authorized owner AI row in the app shell', (tester) async {
    const out = String.fromEnvironment('CAPTURE_PATH');
    const after = bool.fromEnvironment('CAPTURE_AFTER');
    expect(out, isNotEmpty);
    for (final (family, path) in [
      ('Ahem', const String.fromEnvironment('CAPTURE_CJK_FONT')),
      ('MaterialIcons', const String.fromEnvironment('CAPTURE_ICON_FONT')),
    ]) {
      final font = ByteData.sublistView(File(path).readAsBytesSync());
      await (FontLoader(family)..addFont(Future.value(font))).load();
    }
    tester.view.devicePixelRatio = 1;
    tester.view.physicalSize = const Size(390, 844);
    addTearDown(tester.view.resetDevicePixelRatio);
    addTearDown(tester.view.resetPhysicalSize);
    final model = OwnerChatFixture();
    final boundary = GlobalKey();
    await tester.pumpWidget(RepaintBoundary(key: boundary, child: WhaleChatApp(model: model)));
    await tester.tap(find.byTooltip('设置'));
    await tester.pumpAndSettle();
    final anchor = find.byKey(const Key('own-ai-direct-refresh-ai-owner'));
    await tester.scrollUntilVisible(anchor, 260);
    await Scrollable.ensureVisible(tester.element(anchor), alignment: .75);
    await tester.pumpAndSettle();
    expect(find.text('私聊授权：已开启'), findsOneWidget);
    expect(find.byKey(entry), after ? findsOneWidget : findsNothing);
    expect(tester.takeException(), isNull);

    Future<void> capture(String path) async {
      final painter = boundary.currentContext!.findRenderObject()! as RenderRepaintBoundary;
      final encoded = await tester.runAsync(() async {
        final image = await painter.toImage();
        try {
          return await image.toByteData(format: ui.ImageByteFormat.png);
        } finally {
          image.dispose();
        }
      });
      File(path).writeAsBytesSync(encoded!.buffer.asUint8List());
    }

    await capture('$out.png');
    if (!after) return;
    await tester.ensureVisible(find.byKey(entry));
    await tester.tap(find.byKey(entry));
    await tester.pumpAndSettle();
    expect(model.selections, [directId]);
    expect(find.byKey(const Key('composer')), findsOneWidget);
    expect(model.writes, 0);
    expect(model.sends, 0);
    expect(tester.takeException(), isNull);
    await capture('$out-open.png');
  });
}
